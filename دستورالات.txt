🛡️ دستور «همه‌کاره‌ی امن» (توصیه‌شده)

اگر مطمئن نیستی کدام را لازم داری، این را اجرا کن:
powershell

Write-Host "🧹 پاک‌سازی کش..." -ForegroundColor Cyan

# ۱. بستن VS Code
Get-Process Code -ErrorAction SilentlyContinue | Stop-Process -Force -ErrorAction SilentlyContinue
Write-Host "  ✓ VS Code بسته شد" -ForegroundColor Gray

# ۲. حذف __pycache__
$count = (Get-ChildItem -Recurse -Directory -Filter "__pycache__" -ErrorAction SilentlyContinue | Measure-Object).Count
Get-ChildItem -Recurse -Directory -Filter "__pycache__" -ErrorAction SilentlyContinue | Remove-Item -Recurse -Force -ErrorAction SilentlyContinue
Write-Host "  ✓ $count پوشه‌ی __pycache__ حذف شد" -ForegroundColor Gray

# ۳. حذف کش ابزارها
Remove-Item -Recurse -Force ".pytest_cache",".mypy_cache",".ruff_cache" -ErrorAction SilentlyContinue
Write-Host "  ✓ کش ابزارهای توسعه حذف شد" -ForegroundColor Gray

# ۴. حذف کش Pylance
Remove-Item -Recurse -Force "$env:APPDATA\Code\User\workspaceStorage" -ErrorAction SilentlyContinue
Write-Host "  ✓ کش Pylance پاک شد" -ForegroundColor Gray

# ۵. حذف کش عمومی VS Code
Remove-Item -Recurse -Force "$env:APPDATA\Code\Cache","$env:APPDATA\Code\CachedData" -ErrorAction SilentlyContinue
Write-Host "  ✓ کش عمومی VS Code پاک شد" -ForegroundColor Gray

Write-Host ""
Write-Host "✅ تمام! حالا VS Code رو باز کن:" -ForegroundColor Green
Write-Host "   code ." -ForegroundColor Yellow
